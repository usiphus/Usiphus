class FetchError(Exception): pass
class _RateLimiter:
    def __init__(self, rps): self.min_interval = 1.0 / max(rps, 0.1); self.lock = threading.Lock(); self.last = 0.0
    def wait(self):
        with self.lock:
            now = time.time(); delta = self.last + self.min_interval - now
            if delta > 0: time.sleep(delta)
            self.last = time.time()

NCBI_API_KEY = ""
_NCBI_LIMITER = _RateLimiter(3)

def http_get(url, retries=4, timeout=30, accept=None):
    last = ""; headers = {"User-Agent": UA}
    if accept: headers["Accept"] = accept
    for i in range(retries):
        try:
            req = urllib.request.Request(url, headers=headers)
            with urllib.request.urlopen(req, timeout=timeout) as r: return r.read()
        except urllib.error.HTTPError as e:
            last = "HTTP " + str(e.code)
            if e.code in (400, 401, 403, 404, 410): raise FetchError(last + " " + url)
            if e.code == 429: time.sleep(min(30 * (i + 1), 120)); continue
        except Exception as e: last = type(e).__name__
        time.sleep(min(2 * 2 ** i, 30))
    raise FetchError(last + " " + url)

def ncbi_get(endpoint, params):
    _NCBI_LIMITER.wait(); p = dict(params)
    if NCBI_API_KEY: p["api_key"] = NCBI_API_KEY
    p.setdefault("tool", "specialty-harvester"); p.setdefault("email", "research@example.org")
    return http_get(NCBI + "/" + endpoint + "?" + urllib.parse.urlencode(p))

@dataclass
class Candidate:
    source: str; title: str = ""; abstract: str = ""; doi: str = ""; pmcid: str = ""; year: str = ""; journal: str = ""; sections: list = field(default_factory=list)
    def dedup_key(self):
        if self.pmcid: return "pmc:" + self.pmcid.lower()
        if self.doi: return "doi:" + self.doi.lower()
        if self.source.startswith("exam:"): return "exam:" + hashlib.sha1((self.title or "").encode()).hexdigest()[:12]
        return "title:" + re.sub(r"\W+", "", (self.title or "").lower())[:120]
    def folder_name(self):
        if self.pmcid: return self.pmcid
        if self.doi: return "doi_" + re.sub(r"[^a-z0-9]+", "_", self.doi.lower()).strip("_")
        if self.source.startswith("exam:"): return "exam_" + hashlib.sha1((self.title or "").encode()).hexdigest()[:12]
        return None

class SourceAdapter:
    name = ""
    def search(self, key, args): raise NotImplementedError
    def count(self, key, args): raise NotImplementedError

class EuropePMCSource(SourceAdapter):
    name = "europepmc"; YEAR_SPAN = 3
    def _fetch_window(self, key, yf, yt, args):
        out = []; q = query_for(key, yf, yt); cursor = "*"; pages = 0
        while True:
            params = urllib.parse.urlencode({"query": q, "format": "json", "resultType": "lite", "pageSize": 1000, "cursorMark": cursor})
            try: data = json.loads(http_get(EPMC + "/search?" + params))
            except (FetchError, json.JSONDecodeError): break
            pages += 1; results = data.get("resultList", {}).get("result", [])
            for r in results:
                pmcid = r.get("pmcid")
                if not pmcid: continue
                out.append(Candidate(source=self.name, title=r.get("title", "") or "", doi=(r.get("doi") or "").lower(), pmcid=pmcid, year=str(r.get("pubYear") or ""), journal=r.get("journalTitle") or ""))
            if args.max_per_source and len(out) >= args.max_per_source: break
            nxt = data.get("nextCursorMark")
            if not results or not nxt or nxt == cursor: break
            cursor = nxt
        return out
    def search(self, key, args):
        if args.year_from or args.year_to:
            for c in self._fetch_window(key, args.year_from, args.year_to, args): yield c
            return
        try:
            params = urllib.parse.urlencode({"query": query_for(key, None, None), "format": "json", "resultType": "lite", "pageSize": 1})
            total = json.loads(http_get(EPMC + "/search?" + params))["hitCount"]
        except (FetchError, json.JSONDecodeError): total = 0
        if total <= 3000:
            for c in self._fetch_window(key, None, None, args): yield c
            return
        import datetime
        end_year = datetime.datetime.now().year + 1; start_year = 1990; windows = []
        y = start_year
        while y <= end_year: windows.append((y, min(y + self.YEAR_SPAN - 1, end_year))); y += self.YEAR_SPAN
        all_cands = []
        with ThreadPoolExecutor(max_workers=4) as ex:
            futs = {ex.submit(self._fetch_window, key, yf, yt, args): (yf, yt) for yf, yt in windows}
            for f in as_completed(futs):
                try: all_cands.extend(f.result())
                except Exception: pass
        if args.max_per_source: all_cands = all_cands[:args.max_per_source]
        for c in all_cands: yield c
    def count(self, key, args):
        try:
            params = urllib.parse.urlencode({"query": query_for(key, args.year_from, args.year_to), "format": "json", "resultType": "lite", "pageSize": 1})
            return json.loads(http_get(EPMC + "/search?" + params))["hitCount"]
        except (FetchError, json.JSONDecodeError): return 0

def query_for(key, yf, yt):
    _, meshes, kws = SPECIALTIES[key]
    parts = [f'MESH:"{m}"' for m in meshes] + [f'TITLE:"{k}"' if " " in k else f"TITLE:{k}" for k in kws]
    q = "(" + " OR ".join(parts) + ") AND " + BASE
    if yf or yt: q += f" AND (PUB_YEAR:[{yf or 1900} TO {yt or 2100}])"
    return q
