# ---------------------------------------------------------------- helpers
def local(tag): return tag.rsplit("}", 1)[-1] if isinstance(tag, str) else ""
def norm(s): return re.sub(r"\s+", " ", s or "").strip()
def clean_ws(s):
    if not s: return ""
    s = re.sub(r"\s+", " ", s); s = re.sub(r"\s+([,.;:!?])", r"\1", s)
    return s.strip()
def clip(s, n):
    s = norm(s)
    if len(s) <= n: return s
    cut = s[:n]; k = max(cut.rfind(". "), cut.rfind("; "))
    cut = cut[:k + 1] if k > n * 0.5 else cut.rsplit(" ", 1)[0]
    return cut.rstrip(",;:") + " …"
def esc(s): return html.escape(s or "", quote=True) # quote=True prevents attribute injection
def cloze_safe(s): return (s or "").replace("}}", "} }").replace("{{", "{ {")
def stable_id(s): return int(hashlib.sha1(s.encode()).hexdigest()[:8], 16)
def sanitize_tag(s): return re.sub(r"[^\w\-]+", "_", s or "").strip("_").lower()
def clip_for_front(text, n=500): return clip(text, n) if text else ""

def enrich(text):
    text = text or ""
    if not text: return ""
    spans = []
    for cls, pat in HIGHLIGHT_PATTERNS:
        for m in pat.finditer(text):
            if m.start() == m.end(): continue
            spans.append((m.start(), m.end(), cls))
    if not spans: return esc(text)
    spans.sort(key=lambda t: (t[0], -(t[1] - t[0])))
    resolved = []; last_end = -1
    for s, e, cls in spans:
        if s < last_end: continue
        resolved.append((s, e, cls)); last_end = e
    out = []; last = 0
    for s, e, cls in resolved:
        if s > last: out.append(esc(text[last:s]))
        out.append('<span class="' + cls + '">' + esc(text[s:e]) + '</span>')
        last = e
    if last < len(text): out.append(esc(text[last:]))
    return "".join(out)

def flatten(sec, out):
    t = child(sec, "title")
    title = re.sub(r"^[\d.\s]+", "", text_of(t) if t is not None else "").strip(" :.-")
    paras = [text_of(p) for p in sec if local(p.tag) == "p"]
    paras = [p for p in paras if p]
    if paras: out.append((title, " ".join(paras)))
    for s in sec:
        if local(s.tag) == "sec": flatten(s, out)

def pick(secs, pattern):
    got = [txt for title, txt in secs if re.search(pattern, title or "", re.I)]
    return " ".join(got)

def parse_jats(xml_path):
    root = ET.parse(xml_path).getroot(); folder = xml_path.parent; pmcid = folder.name
    meta = {}
    meta_file = folder / "meta.json"
    if meta_file.exists():
        try: meta = json.loads(meta_file.read_text())
        except Exception: pass
    title = meta.get("title") or ""; doi = meta.get("doi") or ""; year = str(meta.get("year") or ""); journal = meta.get("journal") or ""
    for el in root.iter():
        n = local(el.tag)
        if n == "article-title" and not title: title = text_of(el)
        elif n == "article-id" and el.get("pub-id-type") == "doi" and not doi: doi = norm(el.text)
        elif n == "journal-title" and not journal: journal = text_of(el)
        elif n == "year" and not year: year = norm(el.text)
    abs_secs = []
    for ab in root.iter():
        if local(ab.tag) == "abstract" and ab.get("abstract-type") not in ("teaser", "toc"): flatten(ab, abs_secs)
    body = next((e for e in root.iter() if local(e.tag) == "body"), None); secs = []
    if body is not None: flatten(body, secs)
    sec = {k: clean_ws(clip(pick(secs, pat), 5000)) for k, pat in SECTION_PATTERNS.items()}
    abstract_case = clean_ws(pick(abs_secs, r"case summary|case presentation") or " ".join(t for _, t in abs_secs))
    case_text = clean_ws(" ".join(t for title_, t in secs if not re.search(r"introduc|background|discussion|conclusion|reference", title_, re.I)))
    return {"id": pmcid, "title": title, "doi": doi, "year": year, "journal": journal or "", "sec": sec, "abstract_case": abstract_case, "case_text": case_text, "discovered_via": meta.get("discovered_via", ""), "jats_source": meta.get("jats_source", "")}

DX_CUE = re.compile(r"(?:final|definitive|overall|confirmed|ultimate|postoperative|pathological|clinical|primary|secondary|presumptive|provisional|working|initial)?\s*diagnos(?:is|es)\s+(?:of|was|is|were)\s+(?:an?\s+|the\s+)?([A-Za-z][^.;:\n]{2,150})", re.I)
DX_CUE_ALT = re.compile(r"(?:diagnosed\s+(?:with|as(?:\s+(?:having|being))?|to\s+have|to\s+be)|confirmed\s+(?:as|to\s+be|with)|presented\s+with)\s+(?:an?\s+|the\s+)?([A-Za-z][^.;:\n]{2,150})", re.I)
DX_STOP = re.compile(r"\s+(?:was|were|is|are|has|had|have|based|according|by|through|using|after|following|because|due\s+to|which|that|who|on\s+the\s+basis|in\s+view|during|and|via|from|on|at|when|as\s+a\s+result|given|treated|managed|admitted|underwent)\b|[,;(]", re.I)
_FILLER = {"the", "a", "an", "this", "that", "and", "or", "with", "without", "from", "patient", "man", "woman", "child", "infant", "neonate", "year", "month", "day", "week", "old", "case", "report", "presentation", "diagnosis", "made", "was", "were", "is", "are", "bilateral", "unilateral", "left", "right"}

def _plausible(dx):
    if not dx or not (3 <= len(dx) <= 120): return False
    dx_l = dx.lower().strip()
    for prefix in ("classic ", "typical ", "unusual ", "rare presentation", "features of ", "signs of ", "symptoms of ", "clinical features", "clinical signs", "clinical symptoms", "presentation of ", "manifestation of ", "complaints of ", "complaint of ", "history of ", "past history of ", "with complaints of ", "with a history of ", "in the setting of ", "in the context of ", "subsequent to ", "prior to ", "referral for "):
        if dx_l.startswith(prefix): return False
    if re.match(r"^(?:such|these|those|this|that|it|they|we|he|she|you|i)\b", dx_l): return False
    if re.match(r"^(?:injury|trauma|pain|symptoms?|signs?|findings?|features?)\s+(?:to|in|of)\s+(?:her|his|their|the patient)\b", dx_l): return False
    if re.search(r"\b(?:may be|can be|could be|might be|should be|is harmful|is safe|is important|is effective|is associated|can cause|may cause|could cause|should not be|must be|was successful|is controversial|not always|is never|is not always)\b", dx_l): return False
    words = re.findall(r"[A-Za-z][A-Za-z\-]+", dx_l)
    return any(len(w) >= 5 and w not in _FILLER for w in words)

def tidy_dx(s):
    s = norm(s)
    s = re.sub(r"^(?:(?:the|a|an|final|definitive|confirmed|clinical|overall|provisional|presumptive|working|initial|primary|secondary)\s+)+", "", s, flags=re.I)
    s = re.sub(r"^(?:diagnosis|diagnoses)\s*[:\-]?\s*", "", s, flags=re.I)
    s = re.sub(r"^(?:was|were|is|are|of|made|established|confirmed|determined|reached|based\s+on)\s+", "", s, flags=re.I)
    s = s.strip(" .;:-–—")
    m = DX_STOP.search(s)
    if m and m.start() >= 4: s = s[:m.start()].strip(" .;:-–—")
    if s and s[0].islower(): s = s[0].upper() + s[1:]
    return s

def dx_from_text(text):
    if not text: return ""
    for m in DX_CUE.finditer(text):
        d = tidy_dx(m.group(1))
        if _plausible(d): return d
    for m in DX_CUE_ALT.finditer(text):
        d = tidy_dx(m.group(1))
        if _plausible(d): return d
    return ""

def tidy_title_dx(title):
    if not title: return ""
    s = title.strip()
    s = re.sub(r"^a?\s*(?:brief\s+)?case\s+(?:report|study|of)\s*[:\-]?\s*", "", s, flags=re.I)
    s = re.sub(r":\s*.+$", "", s)
    s = re.sub(r",?\s+in\s+(?:an?\s+)?\d+[- ](?:year|month|day|week)s?[- ]old\b.*$", "", s, flags=re.I)
    s = re.sub(r"\s+in\s+(?:an?\s+)?(?:patient|man|woman|boy|girl|child|infant|neonate|adult|elderly)\b.*$", "", s, flags=re.I)
    s = re.sub(r"\s+(?:presenting|manifesting|masquerading)\s+(?:as|with)\s+.*$", "", s, flags=re.I)
    s = re.sub(r"\s+as\s+the\s+(?:initial|first|primary)\s+(?:presentation|manifestation)\s+of\s+.*$", "", s, flags=re.I)
    s = re.sub(r"\s+secondary\s+to\s+.*$", "", s, flags=re.I)
    s = re.sub(r"\s+following\s+.*$", "", s, flags=re.I)
    s = s.strip(" .,;:")
    if len(s) > 80:
        m = re.search(r"\s+in\s+([A-Z][a-zA-Z]+(?:\s+[A-Za-z]+){0,5})$", s)
        if m:
            candidate = m.group(1).strip()
            if 5 < len(candidate) < 80: return candidate
        return ""
    return s if 3 <= len(s) <= 100 else ""

def find_dx(case):
    for key in ("dx", "discussion", "presentation", "chief"):
        d = dx_from_text(case["sec"].get(key, ""))
        if d and not _is_comorbid_only(d, case): return d, key
    for txt in (case.get("abstract_case"), case.get("case_text")):
        d = dx_from_text(txt)
        if d and not _is_comorbid_only(d, case): return d, "text"
    title_clean = tidy_title_dx(case.get("title", ""))
    if title_clean and 3 <= len(title_clean) <= 100: return title_clean, "title"
    return "", "title"

def _is_comorbid_only(d, case):
    if not d or len(d.split()) > 1 or len(d) > 20: return False
    d_l = d.lower()
    comorbid = {"depression", "anxiety", "schizophrenia", "bipolar", "psychosis", "mania", "ptsd", "ocd", "dementia", "hypertension", "diabetes", "asthma", "copd", "cad", "chf", "af", "afib", "ckd", "cirrhosis", "hypothyroidism", "hyperlipidemia", "obesity", "smoking", "alcoholism", "stroke", "tia", "seizure", "epilepsy", "migraine", "osteoporosis", "tobacco", "alcohol", "sle", "lupus", "sjogren", "scleroderma", "ra", "anticoagulation", "chemotherapy", "radiation"}
    if d_l not in comorbid: return False
    return d_l not in (case.get("title", "") or "").lower()

def mask(text, dx):
    if not text or not dx: return text or ""
    dx = dx.strip(" .,;:-–—()")
    if not (4 <= len(dx) <= 100) or not _plausible(dx): return text
    pat = r"(?<!\w)" + re.escape(dx) + r"(?!\w)"
    return re.sub(pat, "[…]", text, flags=re.I)

def build_tags(case, dx, specialty, discovered_via="", card_type="diag", confidence=0.0):
    tags = ["case-report", "card::" + card_type]
    if discovered_via.startswith("exam:"):
        tags.append("exam"); tags.append("source::" + sanitize_tag(discovered_via.split(":", 1)[1]))
    else: tags.append("journal")
    if specialty: tags.append(specialty)
    if case.get("year"): tags.append("year::" + sanitize_tag(case["year"]))
    if case.get("journal"): tags.append("journal::" + sanitize_tag(case["journal"]))
    if confidence >= 0.7: tags.append("conf::high")
    elif confidence < 0.4: tags.append("conf::low")
    else: tags.append("conf::med")
    if specialty == "ophthalmology":
        blob = " ".join([case.get("title", ""), dx or "", case.get("abstract_case", "")[:600]])
        for topic, pats in _SUB.items():
            if any(p.search(blob) for p in pats): tags.append("topic::" + topic)
    seen, out = set(), []
    for t in tags:
        if t not in seen: seen.add(t); out.append(t)
    return out

# ---------------------------------------------------------------- extraction (Pearls, Diff, Triggers)
def extract_pearls(case):
    discussion = case["sec"].get("discussion", "") or ""
    pearls = []
    if discussion:
        m = re.search(r"(?:learning points?|key points?|pearls?|take[- ]home(?:\s+message)?s?|teaching points?|lessons?\s+learned|practice points?|clinical pearls?)\s*[:\-]\s*(.+?)(?:\n\n|\Z)", discussion, re.I | re.S)
        if m:
            bullets = re.split(r"(?:^|\n)\s*(?:[\u2022\u00b7\-\*]|\d+[\.\)])\s+", m.group(1))
            pearls = [clean_ws(b) for b in bullets if 20 < len(clean_ws(b)) < 400]
            if pearls: return pearls[:3]
    if discussion:
        for s in re.split(r"(?<=[.!?])\s+", discussion):
            s = clean_ws(s)
            if len(s) < 30 or len(s) > 300: continue
            if re.search(r"\b(?:important|key|note that|remember|should|must|consider|typically|characteristic|hallmark|classic|pathognomonic|in summary|overall|rule out|workup|management)\b", s, re.I):
                pearls.append(s)
        if pearls: return pearls[:3]
    conclusion = case["sec"].get("conclusion", "") or ""
    if conclusion:
        for s in re.split(r"(?<=[.!?])\s+", conclusion):
            s = clean_ws(s)
            if 40 < len(s) < 350: pearls.append(s)
    return pearls[:3]

DIFF_RE = re.compile(r"\b(?:differential(?: diagnosis)?|mimic(?:s|king)?|masquerad\w*|distinguish(?:ed|ing)? (?:from|between)|differentiat\w+ from|confused with|unlike|whereas|as opposed to|should be differentiated)\b", re.I)

def extract_differential(case):
    text = case["sec"].get("discussion", "") or ""
    if not text: text = case.get("abstract_case") or ""
    hits = []
    for sent in re.split(r"(?<=[.!?])\s+", text):
        sent = clean_ws(sent)
        if 30 < len(sent) < 350 and DIFF_RE.search(sent): hits.append(sent)
    return hits[:2]

def _is_negated_finding(text, match_start, window=60):
    prefix = text[max(0, match_start - window):match_start]
    for boundary in (",", ";", " but ", " however ", " although ", " whereas ", " yet "):
        idx = prefix.lower().rfind(boundary)
        if idx >= 0: prefix = prefix[idx + len(boundary):]
    return bool(re.search(r"\b(?:no|not|without|absent|denies|denied|negative|absence|ruled out|excluded)\b", prefix.lower()))

def _is_followup_sentence(s):
    return bool(re.search(r"\b(?:at the end of|following (?:treatment|surgery|laser|therapy|administration|initiation|completion)|\d+ (?:months?|weeks?|years?) (?:after|following|post)|postoperatively|post-treatment|post-?op|(?:improved|normalized|resolved|increased) to|returned to normal|recovery was|on follow-?up|during follow-?up)\b", s, re.I))

def _partial_dx_leak(s, dx):
    if not dx or not s: return False
    dx_words = set(w for w in re.findall(r"[a-z]{5,}", dx.lower()) if w not in {"with", "without", "from", "after", "following", "patient", "bilateral", "unilateral", "subsequent", "secondary", "primary", "associated"})
    if len(dx_words) < 2: return False
    return len(dx_words & set(re.findall(r"[a-z]{5,}", s.lower()))) >= 2

def _score_trigger_sentence(s, dx_l):
    if not s or (dx_l and dx_l in s.lower()): return 0
    if _is_followup_sentence(s) or _partial_dx_leak(s, dx_l): return 0
    if re.search(r"\b(?:was administered|was prescribed|was treated|was given|was considered as|was diagnosed as|was admitted for|was referred to|had no explanation|no explanations for|was initiated on|had been taking|started on|treated with|managed with|underwent|referred for|evaluation of|workup for|assessment for|admitted for|history of|diagnosed with|treated for|referred to|presented for)\b", s, re.I): return 0
    finding_count = sum(1 for cls, pat in HIGHLIGHT_PATTERNS if cls == "hl-finding" for m in pat.finditer(s) if not _is_negated_finding(s, m.start()))
    return 5 if finding_count > 0 else 0

def extract_trigger_sentence(case, dx):
    dx_l = (dx or "").lower().strip()
    best, best_score = "", 0
    sources = [case["sec"].get(k) or "" for k in ("exam", "presentation", "workup", "lab", "imaging")]
    sources.extend([case.get("abstract_case") or "", case.get("case_text") or ""])
    for text in sources:
        if not text: continue
        for s in re.split(r"(?<=[.!?])\s+", text):
            s = clean_ws(s)
            if not (25 < len(s) < 240): continue
            score = _score_trigger_sentence(s, dx_l)
            if score > best_score: best_score, best = score, s
    return best if best_score >= 3 else ""

def extract_next_step(case, dx):
    dx_l = (dx or "").lower().strip()
    dx_words = set(w for w in re.findall(r"[a-z]{4,}", dx_l))
    for source in ("tx", "outcome", "discussion", "conclusion"):
        text = case["sec"].get(source) or ""
        if not text: continue
        candidates = []
        for s in re.split(r"(?<=[.!?])\s+", text):
            s = clean_ws(s)
            if not (25 < len(s) < 300) or (dx_l and dx_l in s.lower()) or len(dx_words & set(re.findall(r"[a-z]{4,}", s.lower()))) >= 2: continue
            if re.search(r"\b(?:treated with|management|initiated|started on|administered|underwent|performed|next step|immediate|urgent|emergency|referred|advised|recommended)\b", s, re.I): candidates.append((2, s))
            else: candidates.append((1, s))
        if candidates:
            candidates.sort(key=lambda x: -x[0])
            return candidates[0][1]
    return ""

def _diagnosis_confidence(case, dx, dx_source=""):
    if not dx: return 0.0
    dx_l = dx.lower().strip()
    dx_words = set(w for w in re.findall(r"[a-z]{4,}", dx_l))
    if not dx_words: return 0.0
    score = 0.30 if dx_source in ("dx", "discussion", "presentation", "chief", "tx", "outcome", "conclusion", "section") else (0.15 if dx_source == "text" else 0.0)
    title_l = (case.get("title", "") or "").lower()
    if dx_l in title_l: score += 0.4
    else:
        overlap = len(dx_words & set(re.findall(r"[a-z]{4,}", title_l))) / max(1, len(dx_words))
        if overlap >= 0.6: score += 0.3
        elif overlap >= 0.3: score += 0.1
    n_sections = sum(1 for k in ("dx", "discussion", "presentation", "tx", "outcome", "conclusion") if dx_l in (case["sec"].get(k, "") or "").lower())
    if dx_l in (case.get("abstract_case", "") or "").lower(): n_sections += 1
    score += min(0.3, n_sections * 0.1)
    return min(1.0, score)

# ---------------------------------------------------------------- rendering primitives
def _section(key, body_html, variant="exam"):
    if not body_html or not body_html.strip(): return ""
    return '<div class="section-block sec-' + key + ' variant-' + variant + '"><div class="section-label">' + esc(SECTION_LABELS.get(key, key.capitalize())) + '</div><div class="section-body">' + body_html + '</div></div>'

def build_header(case, dx, is_exam, mask_dx=""):
    # Best Practice: Remove the title to prevent answer leaks. Only render chips.
    chips = []
    if is_exam:
        src = case.get("discovered_via", "").replace("exam:", "").split(":", 1)[-1]
        if src: chips.append(("exam", src.replace("_", " ").title()))
    blob = (case.get("title", "") + " " + (dx or "")).lower()
    for topic, pats in _SUB.items():
        if len(chips) >= 3: break
        for pat in pats:
            if pat.search(blob): chips.append(("topic", topic.replace("-", " ").title())); break
    if not chips: return ""
    return '<div class="case-header"><div class="chips">' + "".join('<span class="chip chip-' + kind + '">' + esc(text) + '</span>' for kind, text in chips[:3]) + '</div></div>'

PROMPT_VARIANTS = ["What is the most likely diagnosis?", "Which diagnosis best explains these findings?", "What is the single most likely diagnosis?"]
def pick_prompt(case): return PROMPT_VARIANTS[stable_id(case["id"] + "prompt") % len(PROMPT_VARIANTS)]

def build_diagnosis_front(case, mask_dx, dx, is_exam):
    parts = [build_header(case, dx, is_exam, mask_dx)]
    variant = "exam" if is_exam else "journal"
    sections = []
    used = set()
    
    # Best Practice: Limit to 3 sentences (~500 chars) to avoid overwhelming the user.
    for k in ("chief", "hpi", "exam"):
        if len(sections) >= 2: break
        txt = case["sec"].get(k)
        if txt and k not in used:
            body = mask(txt, mask_dx) if mask_dx else txt
            sections.append(_section(k, enrich(clip_for_front(body, 250)), variant))
            used.add(k)
            
    if not sections:
        # Fallback to presentation or abstract
        raw = case["sec"].get("presentation") or case["sec"].get("case") or case.get("abstract_case") or ""
        if raw:
            body = mask(raw, mask_dx) if mask_dx else raw
            sections.append(_section("presentation", enrich(clip_for_front(body, 500)), variant))
            
    parts.extend(sections)
    prompt = pick_prompt(case)
    # Best Practice: The cloze should be standard syntax. No green box needed later.
    parts.append('<div class="question-box"><div class="question-label">' + esc(prompt) + '</div><div class="question-cloze">{{c1::' + esc(cloze_safe(dx)) + '}}</div></div>')
    return "".join(parts)

def build_trigger_front(case, trigger, dx, is_exam):
    parts = [build_header(case, dx, is_exam)]
    parts.append('<div class="trigger-card"><div class="trigger-label">Key finding</div><div class="trigger-body">' + enrich(trigger) + '</div></div>')
    parts.append('<div class="question-box"><div class="question-label">Which diagnosis does this finding suggest?</div><div class="question-cloze">{{c1::' + esc(cloze_safe(dx)) + '}}</div></div>')
    return "".join(parts)

def build_next_step_front(case, mask_dx, dx, next_step, is_exam):
    raw = (case["sec"].get("presentation") or case["sec"].get("chief") or case.get("abstract_case") or case.get("case_text") or "")
    if mask_dx: raw = mask(raw, mask_dx)
    parts = [build_header(case, dx, is_exam)]
    if raw: parts.append(_section("presentation", enrich(clip_for_front(raw, 500)), "exam" if is_exam else "journal"))
    parts.append('<div class="question-box"><div class="question-label">What is the best immediate next step?</div><div class="question-cloze">{{c1::' + esc(cloze_safe(next_step)) + '}}</div></div>')
    return "".join(parts)

def build_source_block(case):
    lines = []
    if case.get("title"): lines.append('<div class="src-title">' + esc(case["title"]) + '</div>')
    meta = []
    if case.get("journal"): meta.append(esc(case["journal"]))
    if case.get("year"): meta.append(esc(case["year"]))
    if meta: lines.append('<div class="src-meta">' + " · ".join(meta) + '</div>')
    if case.get("doi"):
        url = case["doi"] if case["doi"].startswith("http") else "https://doi.org/" + case["doi"]
        lines.append('<div class="src-link"><a href="' + esc(url) + '">' + esc(case["doi"]) + '</a></div>')
    if case.get("id") and str(case["id"]).upper().startswith("PMC"):
        lines.append('<div class="src-link"><a href="https://www.ncbi.nlm.nih.gov/pmc/articles/' + esc(case["id"]) + '/">' + esc(case["id"]) + '</a></div>')
    return '<div class="source">' + "".join(lines) + '</div>'

def build_back(case, dx, dx_source, is_exam):
    # Best Practice: Do NOT include the answer in a green box. The cloze handles the reveal.
    variant = "exam-back" if is_exam else "journal-back"
    parts = []
    
    pearls = extract_pearls(case)
    if pearls:
        cls = "pearls-box" + ("" if is_exam else " pearls-journal")
        parts.append('<div class="' + cls + '"><div class="pearls-label">Key pearls</div><ul>' + "".join('<li>' + enrich(p) + '</li>' for p in pearls) + '</ul></div>')
        
    diffs = extract_differential(case)
    if diffs:
        parts.append('<div class="diff-box"><div class="diff-label">Differential</div><ul>' + "".join('<li>' + enrich(d) + '</li>' for d in diffs) + '</ul></div>')
        
    parts.append(build_source_block(case))
    
    full = []
    full_order = EXAM_FULL_ORDER if is_exam else JOURNAL_FULL_ORDER
    for k in full_order:
        txt = case["sec"].get(k)
        if not txt: continue
        if k in ("discussion", "presentation") and len(txt) > 1000: txt = clip_for_front(txt, 1000)
        section = _section(k, enrich(txt), variant)
        if section: full.append(section)
        
    if full:
        parts.append('<details class="more"><summary>Show full case text</summary>' + "".join(full) + '</details>')
    return "".join(parts)

def main():
    ap = argparse.ArgumentParser(description="Turn case reports into Anki decks.")
    ap.add_argument("input", type=Path)
    ap.add_argument("-o", "--output", default="ophthalmology.apkg")
    ap.add_argument("--deck", default="Ophthalmology")
    ap.add_argument("--specialty", default="ophthalmology")
    ap.add_argument("--text-only", action="store_true")
    ap.add_argument("--limit", type=int, default=0)
    ap.add_argument("--chunk", type=int, default=0)
    ap.add_argument("--no-mask", action="store_true")
    ap.add_argument("--no-trigger", action="store_true")
    ap.add_argument("--no-next-step", action="store_true")
    ap.add_argument("--no-dedup", action="store_true")
    ap.add_argument("--interleave", action="store_true")
    ap.add_argument("--preview", action="store_true")
    ap.add_argument("--quality-report", action="store_true")
    ap.add_argument("--include-quarantine", action="store_true")
    ap.add_argument("--curate", choices=["balanced", "strict", "ultra"], default="")
    args = ap.parse_args()

    if not args.input.exists(): sys.exit("Input not found: " + str(args.input))
    xml_files = list(args.input.rglob("article.xml")) if args.input.is_dir() else [args.input]

    import genanki
    model = genanki.Model(
        1607392324, "Clinical Case Exam v2",
        fields=[{"name": "Front"}, {"name": "Hint"}, {"name": "Back"}],
        templates=[{"name": "Cloze", "qfmt": "{{cloze:Front}}{{#Hint}}<br><br>{{hint:Hint}}{{/Hint}}", "afmt": "{{cloze:Front}}<hr id=answer>{{Back}}"}],
        css=CARD_CSS, model_type=genanki.Model.CLOZE)

    entries = []
    media_files = set()
    report_rows = []
    counts = {"diag": 0, "trig": 0, "next": 0}
    skipped = 0

    for fp_path in xml_files:
        try:
            case = parse_jats(fp_path)
            if args.specialty == "ophthalmology" and not _positive_ophthalmology(case.get("title", ""), case.get("abstract_case", "")): skipped += 1; continue
            dx, dx_source = find_dx(case)
            mask_dx = "" if args.no_mask else dx
            is_exam = case.get("discovered_via", "").startswith("exam:")
            conf = _diagnosis_confidence(case, dx, dx_source)
            diag_front = build_diagnosis_front(case, mask_dx, dx, is_exam)
            
            # Quality gate
            blocking, soft, score = _quality_gate(case, dx, diag_front, dx_source, conf, args.specialty)
            action = _action(blocking, soft, score)
            report_rows.append([case["id"], case["title"][:80], dx, dx_source, ";".join(blocking), ";".join(soft), score, action, round(conf, 2)])
            
            if action == "rejected": skipped += 1; continue
            entries.append({"case": case, "dx": dx, "dx_source": dx_source, "mask_dx": mask_dx, "fp": _fingerprint(case), "action": action, "is_exam": is_exam, "confidence": conf})
            for img in case.get("images", []):
                if img.get("path"): media_files.add(str(img["path"]))
        except Exception as e: print("parse skip " + fp_path.name + ": " + str(e))

    print("Parsed " + str(len(entries)) + " usable cases (" + str(skipped) + " skipped)")

    if args.quality_report:
        Path("decks").mkdir(exist_ok=True)
        with open("decks/quality_report.csv", "w", newline="", encoding="utf-8") as f:
            w = csv.writer(f)
            w.writerow(["pmcid", "title", "dx", "dx_source", "blocking", "soft", "score", "action", "confidence"])
            for row in report_rows: w.writerow(row)
        print("Wrote decks/quality_report.csv (" + str(len(report_rows)) + " rows)")

    if not args.no_dedup:
        before = len(entries)
        entries = _dedupe(entries)
        print("Dedup: " + str(before) + " -> " + str(len(entries)))

    kept_entries = [e for e in entries if e["action"] == "kept"]
    quarantined = [e for e in entries if e["action"] == "quarantined"]

    for entry in kept_entries:
        entry["trigger"] = extract_trigger_sentence(entry["case"], entry["dx"])
        entry["next_step"] = extract_next_step(entry["case"], entry["dx"])
        entry["has_trigger"] = bool(entry["trigger"])
        entry["has_next"] = bool(entry["next_step"])
        entry["curation_score"] = _curation_score(entry["case"], entry["dx"], entry["dx_source"], entry["confidence"], entry["has_trigger"], entry["has_next"])

    if args.curate:
        thresholds = {"balanced": 4, "strict": 6, "ultra": 8}
        threshold = thresholds[args.curate]
        before = len(kept_entries)
        kept_entries = [e for e in kept_entries if e["curation_score"] >= threshold]
        dropped = before - len(kept_entries)
        print("Curation (" + args.curate + ", score>=" + str(threshold) + "): " + str(before) + " -> " + str(len(kept_entries)) + " cases (" + str(dropped) + " dropped)")

    def _make_notes(entry_list):
        notes_out = []
        for entry in entry_list:
            case = entry["case"]; dx = entry["dx"]; dx_source = entry["dx_source"]; mask_dx = entry["mask_dx"]; is_exam = entry["is_exam"]; conf = entry["confidence"]
            back_html = build_back(case, dx, dx_source, is_exam)
            
            diag_front = build_diagnosis_front(case, mask_dx, dx, is_exam)
            chips = [topic.replace("-", " ").title() for topic, pats in _SUB.items() if any(p.search((case.get("title", "") + " " + dx).lower()) for p in pats)]
            hint = "Topic: " + chips[0] if chips else ""

            notes_out.append(genanki.Note(model=model, fields=[diag_front, hint, back_html], guid=genanki.guid_for(case["id"] + str(case["title"]) + "diag"), tags=build_tags(case, dx, args.specialty, case.get("discovered_via", ""), "diag", conf)))
            counts["diag"] += 1

            if not args.no_trigger:
                trigger = entry.get("trigger") or extract_trigger_sentence(case, dx)
                if trigger and len(trigger) > 30:
                    trig_front = build_trigger_front(case, trigger, dx, is_exam)
                    notes_out.append(genanki.Note(model=model, fields=[trig_front, hint, back_html], guid=genanki.guid_for(case["id"] + str(case["title"]) + "trig"), tags=build_tags(case, dx, args.specialty, case.get("discovered_via", ""), "trig", conf)))
                    counts["trig"] += 1

            if not args.no_next_step:
                next_step = entry.get("next_step") or extract_next_step(case, dx)
                if next_step and len(next_step) > 20:
                    nxt_front = build_next_step_front(case, mask_dx, dx, next_step, is_exam)
                    if len(nxt_front) > 100:
                        notes_out.append(genanki.Note(model=model, fields=[nxt_front, "", back_html], guid=genanki.guid_for(case["id"] + str(case["title"]) + "next"), tags=build_tags(case, dx, args.specialty, case.get("discovered_via", ""), "next", conf)))
                        counts["next"] += 1
        return notes_out

    notes = _make_notes(kept_entries)
    quarantine_notes = _make_notes(quarantined) if args.include_quarantine else []

    print("Cards: diag=" + str(counts["diag"]) + " trig=" + str(counts["trig"]) + " next=" + str(counts["next"]) + " total=" + str(len(notes)))
    if counts["diag"]: print("Coverage: trig " + str(int(100 * counts["trig"] / counts["diag"])) + "% | next " + str(int(100 * counts["next"] / counts["diag"])) + "%")

    if args.limit and len(notes) > args.limit: notes = notes[:args.limit]
    if args.interleave: random.Random(42).shuffle(notes); print("Interleaved (seed=42)")

    if args.preview:
        Path("decks").mkdir(exist_ok=True); preview_path = Path("decks/preview.html")
        html_out = ("<html><head><meta charset='utf-8'><style>" + CARD_CSS + "</style></head><body>" + "".join('<div class="card">' + n.fields[0] + '<hr id=answer>' + n.fields[2] + '</div>' for n in notes[:30]) + "</body></html>")
        preview_path.write_text(html_out, encoding="utf-8"); print("Preview: " + str(preview_path))

    Path("decks").mkdir(exist_ok=True)
    if args.chunk:
        for i in range(0, len(notes), args.chunk):
            part = notes[i:i + args.chunk]; chunk_num = (i // args.chunk) + 1
            deck = genanki.Deck(stable_id(args.deck) + chunk_num, args.deck + " - Part " + str(chunk_num).zfill(2))
            for n in part: deck.add_note(n)
            pkg = genanki.Package(deck)
            if not args.text_only and media_files: pkg.media_files = sorted(media_files)
            out = Path("decks") / (Path(args.output).stem + "_part" + str(chunk_num).zfill(2) + ".apkg")
            pkg.write_to_file(str(out)); print("Wrote " + str(len(part)) + " cards to " + str(out))
    else:
        deck = genanki.Deck(stable_id(args.deck), args.deck)
        for n in notes: deck.add_note(n)
        pkg = genanki.Package(deck)
        if not args.text_only and media_files: pkg.media_files = sorted(media_files)
        out = Path(args.output) if Path(args.output).is_absolute() else Path("decks") / args.output
        pkg.write_to_file(str(out)); print("Wrote " + str(len(notes)) + " cards to " + str(out))

    if quarantine_notes:
        qdeck = genanki.Deck(stable_id(args.deck) + 999, args.deck + " - Quarantine")
        for n in quarantine_notes: qdeck.add_note(n)
        qpkg = genanki.Package(qdeck)
        if not args.text
